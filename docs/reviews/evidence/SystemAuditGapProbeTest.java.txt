package com.click.service.overlay;

import com.click.data.snapshot.*;
import java.util.List;
import org.json.JSONObject;
import org.junit.Test;
import static org.junit.Assert.*;

/** Audit probes record current gaps; passing these is NOT an acceptance gate. */
public class SystemAuditGapProbeTest {
 private ScreenSnapshot menu() {
  var list=new AccessibilityNodeSnapshot("list",null,true,"com.android.settings",null,null,null,
    "android.widget.ListView","list",new int[]{0,200,1080,2200},false,true,false,true,false,null,42);
  return new ScreenSnapshot("com.android.settings","2026-11-09T00:00:00Z",1080,2400,List.of(list),42);
 }
 @Test public void expandedStatesRejectTheSameScrollRecoveryAcceptedForLegacyStates() throws Exception {
  var recovery=new JSONObject().put("reason_code","TARGET_NOT_VISIBLE").put("next_action","manual_navigation")
    .put("waiting_for","user").put("recoverable",true);
  var policy=new JSONObject().put("recovery",recovery);
  assertTrue(ObservationRecovery.canSearchByScrolling("S2_CATEGORY",policy,menu()));
  assertFalse(ObservationRecovery.canSearchByScrolling("S_CATEGORY",policy,menu()));
  assertFalse(ObservationRecovery.canSearchByScrolling("S_INTERMEDIATE",policy,menu()));
 }
 @Test public void settingsContainerPathChurnCanResetNoProgressDetectionOneHundredTimes() {
  var search=new AutomaticScrollSearch();
  for(int i=0;i<100;i++) {
   var frame=new NavigationState("node_"+i,NavigationState.Axis.VERTICAL,null,null,-1,-1,-1,"identical-page");
   String scope="episode:S1_SETTINGS_HOME:com.android.settings:42:"+frame.containerId;
   assertEquals(AutomaticScrollSearch.Status.MOVE,search.plan(scope,frame).status);
  }
 }
 @Test public void unrelatedWifiStateStillChangesSettingsNavigationIdentity() {
  var before=menu().withWifiState(2);var after=menu().withWifiState(3);
  assertFalse(ObservationRecovery.sameNavigationScreen(before,after));
 }
 private ScreenSnapshot namedRow(String label,boolean toggle) {
  var row=new AccessibilityNodeSnapshot("target",null,true,"com.android.settings",label,null,
    "android:id/title",toggle?"android.widget.Switch":"android.widget.TextView",toggle?"switch":"menu_item",
    new int[]{20,200,1000,300},true,true,false,false,toggle,null,42);
  return new ScreenSnapshot("com.android.settings","2026-11-09T00:00:00Z",1080,2400,List.of(row),42);
 }
 @Test public void expandedNavigationNamesAreRejectedByTheNativeAutoTapPolicy() {
  var old=namedRow("디스플레이",false);
  assertNotNull(com.click.service.accessibility.TouchAutomationPolicy.clickableTarget(old,"target",old.nodes.get(0).bounds));
  for(String name:new String[]{"연결","블루투스","Wi-Fi","배터리","알림"}) {
   var expanded=namedRow(name,false);
   assertNull(com.click.service.accessibility.TouchAutomationPolicy.clickableTarget(expanded,"target",expanded.nodes.get(0).bounds));
  }
 }
 @Test public void approvedNativeCheckedContractsStillHaveNoNativeAutoTapPermission() {
  var toggle=namedRow("블루투스",true);
  assertNull(com.click.service.accessibility.TouchAutomationPolicy.clickableTarget(toggle,"target",toggle.nodes.get(0).bounds));
 }

}
